    Header set Pragma "no-cache"
    Header set Expires "0"

  Header always set Cache-Control "no-cache, no-store, must-revalidate" env=BP_NOCACHE
</IfModule>

<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /

# Keep WordPress admin, login, and core
RewriteRule ^wp-admin(/.*)?$ - [L]
RewriteRule ^wp-login\.php$ - [L]
RewriteRule ^wp-cron\.php$ - [L]
RewriteRule ^xmlrpc\.php$ - [L]
RewriteCond %{REQUEST_URI} ^/wp-content/ [NC]
RewriteRule .* - [L]
RewriteCond %{REQUEST_URI} ^/wp-includes/ [NC]
RewriteRule .* - [L]

# WordPress REST API (wp-json is not a real folder; WordPress serves it via index.php)
RewriteRule ^wp-json/?$ /index.php?rest_route=/ [QSA,L]
RewriteRule ^wp-json/(.*)$ /index.php?rest_route=/$1 [QSA,L]

# Existing files/folders stay as-is
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# Everything else → React SPA
RewriteRule . /index.html [L]
</IfModule>
